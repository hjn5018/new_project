💡 JUnit 5 단위 테스트 작성 실무 사례 (선착순 수량 검증)

@Test
@DisplayName("재고가 0개일 때 주문 시 StockEmptyException 예외가 발생한다")
void shouldThrowExceptionWhenStockIsEmpty() {
    // Given
    Product product = new Product("PROD-101", 0); // 재고 0개
    
    // When & Then
    assertThatThrownBy(() -> orderService.placeOrder(product, 1))
        .isInstanceOf(StockEmptyException.class)
        .hasMessage("재고가 부족합니다.");
}
